// ─── Results ──────────────────────────────────────────────────────────────────

function ResultsView({ params, onChange, onSearch, onSelect, onOpenMap }: { params: SearchParams; onChange: (p: SearchParams) => void; onSearch: () => void; onSelect: (h: Hotel) => void; onOpenMap?: (hotelId?: string) => void }) {
  const [stars, setStars] = useState<number[]>([]);
  const [maxPrice, setMaxPrice] = useState(250);
  const [classification, setClassification] = useState<string>("all");
  const [parish, setParish] = useState<string>("all");
  const [sortBy, setSortBy] = useState<"price" | "rating">("rating");

  // Filtros de Estancia Populares (Booking/Expedia style)
  const [filterParking, setFilterParking] = useState(false);
  const [filterPetFriendly, setFilterPetFriendly] = useState(false);
  const [filterBreakfast, setFilterBreakfast] = useState(false);
  const [filterPool, setFilterPool] = useState(false);
  const [filterAC, setFilterAC] = useState(false);
  const [filterBeachfront, setFilterBeachfront] = useState(false);

  const nights = nightsBetween(params.checkIn, params.checkOut);
  let results = HOTELS.filter(h => {
    const d = !params.destination || h.city.toLowerCase().includes(params.destination.toLowerCase()) || (h.parish && h.parish.toLowerCase().includes(params.destination.toLowerCase()));
    const s = stars.length === 0 || stars.includes(h.stars);
    const p = h.priceFrom <= maxPrice;
    const c = h.rooms.some(r => r.capacity >= params.adults);
    const cl = classification === "all" || (h.classification && h.classification.toLowerCase() === classification.toLowerCase());
    const pa = parish === "all" || (h.parish && h.parish.toLowerCase() === parish.toLowerCase());

    const matchPark = !filterParking || h.amenities.some(a => a.toLowerCase().includes("parqueadero") || a.toLowerCase().includes("estacionamiento")) || h.tags.some(t => t.toLowerCase().includes("parqueadero"));
    const matchPet = !filterPetFriendly || h.amenities.some(a => a.toLowerCase().includes("mascota") || a.toLowerCase().includes("pet")) || h.tags.some(t => t.toLowerCase().includes("mascota") || t.toLowerCase().includes("pet"));
    const matchBf = !filterBreakfast || h.amenities.some(a => a.toLowerCase().includes("desayuno")) || h.rooms.some(r => r.breakfastIncluded);
    const matchPool = !filterPool || h.amenities.some(a => a.toLowerCase().includes("piscina")) || h.tags.some(t => t.toLowerCase().includes("piscina"));
    const matchAC = !filterAC || h.amenities.some(a => a.toLowerCase().includes("aire") || a.toLowerCase().includes("a/c") || a.toLowerCase().includes("climatizado"));
    const matchBeach = !filterBeachfront || h.tags.some(t => t.toLowerCase().includes("mar") || t.toLowerCase().includes("playa"));

    return d && s && p && c && cl && pa && matchPark && matchPet && matchBf && matchPool && matchAC && matchBeach;
  });
  if (sortBy === "price") results = [...results].sort((a, b) => a.priceFrom - b.priceFrom);
  if (sortBy === "rating") results = [...results].sort((a, b) => b.rating - a.rating);

  return (
    <div style={{ background: "#f8fafc", minHeight: "100vh" }}>
      <div className="sticky top-0 z-20 border-b shadow-sm" style={{ background: "#fff", borderColor: "#e2e8f0" }}>
        <div className="max-w-5xl mx-auto px-4 py-3">
          <SearchBar params={params} onChange={onChange} onSearch={onSearch} variant="sticky" />
        </div>
      </div>
      <div className="max-w-5xl mx-auto px-4 py-6 flex gap-5">
        <aside className="w-56 flex-shrink-0 hidden lg:block space-y-4">
          <div className="rounded-xl p-4 border bg-white" style={{ borderColor: "#e2e8f0" }}>
            <div className="flex items-center justify-between mb-3">
              <h4 className="font-bold text-slate-700 text-sm">Categoría</h4>
              {stars.length > 0 && (
                <button onClick={() => setStars([])} className="text-[11px] text-amber-600 hover:underline">Limpiar</button>
              )}
            </div>
            {[5, 4, 3, 2, 1].map(s => (
              <label key={s} className="flex items-center justify-between gap-1 mb-2 cursor-pointer hover:bg-slate-50 p-1 rounded transition-colors">
                <div className="flex items-center gap-1.5">
                  <input
                    type="checkbox"
                    className="accent-amber-400"
                    checked={stars.includes(s)}
                    onChange={() => setStars(prev => prev.includes(s) ? prev.filter(x => x !== s) : [...prev, s])}
                  />
                  <div className="flex gap-0.5">
                    {Array.from({ length: s }).map((_, i) => <span key={i} style={{ color: "#f59e0b", fontSize: 11 }}>★</span>)}
                    {Array.from({ length: 5 - s }).map((_, i) => <span key={i} style={{ color: "#d1d5db", fontSize: 11 }}>★</span>)}
                  </div>
                  <span className="text-[11px] text-slate-600 font-medium">
                    {s === 5 ? "5★ Lujo" : s === 4 ? "4★ Primera" : s === 3 ? "3★ Superior" : s === 2 ? "2★ Turista" : "1★ Económico"}
                  </span>
                </div>
                <span className="text-[10px] text-slate-400 font-mono">({HOTELS.filter(h => h.stars === s).length})</span>
              </label>
            ))}
          </div>

          {/* Filtros Populares de Estancia (Parking, Pet Friendly, Desayuno, A/C, Piscina) */}
          <div className="rounded-xl p-4 border bg-white" style={{ borderColor: "#e2e8f0" }}>
            <div className="flex items-center justify-between mb-2.5">
              <h4 className="font-bold text-slate-700 text-sm">Filtros Populares</h4>
              {(filterParking || filterPetFriendly || filterBreakfast || filterPool || filterAC || filterBeachfront) && (
                <button
                  type="button"
                  onClick={() => {
                    setFilterParking(false);
                    setFilterPetFriendly(false);
                    setFilterBreakfast(false);
                    setFilterPool(false);
                    setFilterAC(false);
                    setFilterBeachfront(false);
                  }}
                  className="text-[10px] font-bold text-amber-600 hover:underline cursor-pointer"
                >
                  Limpiar
                </button>
              )}
            </div>
            <div className="space-y-1.5 text-xs">
              <label className="flex items-center justify-between gap-1.5 cursor-pointer hover:bg-slate-50 p-1 rounded transition-colors">
                <div className="flex items-center gap-2">
                  <input
                    type="checkbox"
                    checked={filterParking}
                    onChange={e => setFilterParking(e.target.checked)}
                    className="accent-amber-500 rounded cursor-pointer"
                  />
                  <span className="text-slate-700 font-medium">🅿️ Parqueadero gratis</span>
                </div>
              </label>

              <label className="flex items-center justify-between gap-1.5 cursor-pointer hover:bg-slate-50 p-1 rounded transition-colors">
                <div className="flex items-center gap-2">
                  <input
                    type="checkbox"
                    checked={filterPetFriendly}
                    onChange={e => setFilterPetFriendly(e.target.checked)}
                    className="accent-amber-500 rounded cursor-pointer"
                  />
                  <span className="text-slate-700 font-medium">🐾 Admite mascotas</span>
                </div>
              </label>

              <label className="flex items-center justify-between gap-1.5 cursor-pointer hover:bg-slate-50 p-1 rounded transition-colors">
                <div className="flex items-center gap-2">
                  <input
                    type="checkbox"
                    checked={filterBreakfast}
                    onChange={e => setFilterBreakfast(e.target.checked)}
                    className="accent-amber-500 rounded cursor-pointer"
                  />
                  <span className="text-slate-700 font-medium">☕ Desayuno incluido</span>
                </div>
              </label>

              <label className="flex items-center justify-between gap-1.5 cursor-pointer hover:bg-slate-50 p-1 rounded transition-colors">
                <div className="flex items-center gap-2">
                  <input
                    type="checkbox"
                    checked={filterAC}
                    onChange={e => setFilterAC(e.target.checked)}
                    className="accent-amber-500 rounded cursor-pointer"
                  />
                  <span className="text-slate-700 font-medium">❄️ Aire Acondicionado</span>
                </div>
              </label>

              <label className="flex items-center justify-between gap-1.5 cursor-pointer hover:bg-slate-50 p-1 rounded transition-colors">
                <div className="flex items-center gap-2">
                  <input
                    type="checkbox"
                    checked={filterPool}
                    onChange={e => setFilterPool(e.target.checked)}
                    className="accent-amber-500 rounded cursor-pointer"
                  />
                  <span className="text-slate-700 font-medium">🏊 Con Piscina</span>
                </div>
              </label>

              <label className="flex items-center justify-between gap-1.5 cursor-pointer hover:bg-slate-50 p-1 rounded transition-colors">
                <div className="flex items-center gap-2">
                  <input
                    type="checkbox"
                    checked={filterBeachfront}
                    onChange={e => setFilterBeachfront(e.target.checked)}
                    className="accent-amber-500 rounded cursor-pointer"
                  />
                  <span className="text-slate-700 font-medium">🏖️ Frente al Mar / Playa</span>
                </div>
              </label>
            </div>
          </div>

          <div className="rounded-xl p-4 border bg-white" style={{ borderColor: "#e2e8f0" }}>
            <h4 className="font-bold text-slate-700 text-sm mb-2">Clasificación</h4>
            <div className="space-y-1">
              {[
                { id: "all", label: "Todos los tipos", count: HOTELS.length },
                { id: "Hotel", label: "Hotel", count: HOTELS.filter(h => h.classification === "Hotel").length },
                { id: "Hostal", label: "Hostal", count: HOTELS.filter(h => h.classification === "Hostal").length },
                { id: "Hostería", label: "Hostería", count: HOTELS.filter(h => h.classification === "Hostería").length },
              ].map(item => (
                <button
                  key={item.id}
                  onClick={() => setClassification(item.id)}
                  className={`w-full flex items-center justify-between text-xs px-2.5 py-1.5 rounded-lg text-left transition-all ${
                    classification === item.id ? "bg-amber-100 text-amber-900 font-bold" : "text-slate-600 hover:bg-slate-50"
                  }`}
                >
                  <span>{item.label}</span>
                  <span className="font-mono text-[10px] text-slate-400">({item.count})</span>
                </button>
              ))}
            </div>
          </div>

          <div className="rounded-xl p-4 border bg-white" style={{ borderColor: "#e2e8f0" }}>
            <h4 className="font-bold text-slate-700 text-sm mb-2">Parroquia</h4>
            <select
              value={parish}
              onChange={e => setParish(e.target.value)}
              className="w-full text-xs rounded-lg border p-2 bg-slate-50 border-slate-200 text-slate-700 focus:outline-none focus:ring-1 focus:ring-amber-500"
            >
              <option value="all">Todas las parroquias (116)</option>
              {Array.from(new Set(HOTELS.map(h => h.parish).filter(Boolean))).sort().map(p => (
                <option key={p} value={p}>{p} ({HOTELS.filter(h => h.parish === p).length})</option>
              ))}
            </select>
          </div>

          <div className="rounded-xl p-4 border bg-white" style={{ borderColor: "#e2e8f0" }}>
            <div className="flex justify-between items-center mb-2">
              <h4 className="font-bold text-slate-700 text-sm">Precio máx./noche</h4>
              <span className="font-bold text-amber-600 text-sm font-mono">${maxPrice}</span>
            </div>
            <input
              type="range"
              min={25}
              max={300}
              step={5}
              value={maxPrice}
              onChange={e => setMaxPrice(Number(e.target.value))}
              className="w-full accent-amber-500 cursor-pointer"
            />
            <div className="flex justify-between text-[11px] text-slate-400 mt-1">
              <span>$25</span>
              <span>$150</span>
              <span>$300+</span>
            </div>
          </div>
        </aside>
        <div className="flex-1 min-w-0">
          <div className="flex items-center justify-between mb-4 flex-wrap gap-2">
            <div>
              <h2 className="font-bold text-slate-800">{results.length} alojamiento{results.length !== 1 ? "s" : ""}{params.destination ? ` en ${params.destination}` : ""}</h2>
              {nights > 0 && <p className="text-xs text-slate-400">{nights} noche{nights > 1 ? "s" : ""} · {params.adults} adulto{params.adults > 1 ? "s" : ""}</p>}
            </div>
            <div className="flex items-center gap-2 flex-wrap">
              {onOpenMap && (
                <button
                  type="button"
                  onClick={() => onOpenMap()}
                  className="px-3 py-1.5 rounded-lg bg-amber-500 hover:bg-amber-400 text-slate-950 font-bold text-xs flex items-center gap-1.5 shadow-xs transition-all cursor-pointer hover:scale-105"
                  title="Explorar todos los alojamientos en el plano interactivo de Manta"
                >
                  <span>🗺️</span>
                  <span>Ver en Plano ({results.length})</span>
                </button>
              )}
              <div className="flex gap-1.5 text-xs">
                <span className="text-slate-400 self-center">Ordenar:</span>
                {[{ k: "rating" as const, l: "Valoración" }, { k: "price" as const, l: "Precio" }].map(s => (
                  <button key={s.k} onClick={() => setSortBy(s.k)} className="px-3 py-1.5 rounded-lg font-semibold border transition-all" style={{ background: sortBy === s.k ? "#f59e0b" : "#fff", color: sortBy === s.k ? "#0c1220" : "#64748b", borderColor: sortBy === s.k ? "#f59e0b" : "#e2e8f0" }}>{s.l}</button>
                ))}
              </div>
            </div>
          </div>
          {/* Chips de Filtros Rápidos (Mobile & Desktop) */}
          <div className="flex items-center gap-1.5 overflow-x-auto pb-1 mb-3 scrollbar-none text-xs">
            <span className="text-slate-400 text-[11px] font-semibold mr-1 whitespace-nowrap">Filtros:</span>
            {[
              { id: "park", label: "🅿️ Parqueadero", active: filterParking, toggle: () => setFilterParking(!filterParking) },
              { id: "pet", label: "🐾 Pet Friendly", active: filterPetFriendly, toggle: () => setFilterPetFriendly(!filterPetFriendly) },
              { id: "bf", label: "☕ Desayuno", active: filterBreakfast, toggle: () => setFilterBreakfast(!filterBreakfast) },
              { id: "ac", label: "❄️ A/C", active: filterAC, toggle: () => setFilterAC(!filterAC) },
              { id: "pool", label: "🏊 Piscina", active: filterPool, toggle: () => setFilterPool(!filterPool) },
              { id: "beach", label: "🏖️ Frente al mar", active: filterBeachfront, toggle: () => setFilterBeachfront(!filterBeachfront) },
            ].map(chip => (
              <button
                key={chip.id}
                type="button"
                onClick={chip.toggle}
                className={`px-3 py-1 rounded-full text-xs font-semibold whitespace-nowrap transition-all cursor-pointer border ${
                  chip.active
                    ? "bg-slate-900 text-white border-slate-900 shadow-xs"
                    : "bg-white text-slate-600 border-slate-200 hover:bg-slate-100"
                }`}
              >
                {chip.label}
              </button>
            ))}
          </div>

          <div className="space-y-4">
            {results.map(h => <HotelCard key={h.id} hotel={h} nights={nights} onSelect={() => onSelect(h)} onOpenMap={onOpenMap} />)}
            {results.length === 0 && (
              <div className="text-center py-16 rounded-2xl border bg-white" style={{ borderColor: "#e2e8f0" }}>
                <p className="text-4xl mb-3">🏨</p>
                <p className="font-semibold text-slate-700">No encontramos alojamientos con estos filtros.</p>
              </div>
            )}
          </div>
        </div>
      </div>
    </div>
  );
}
